        SET this.transactionData = transactionData
    
    METHOD processPayment()
        PRINT "Processing Payment..."
        PRINT transactionData.getTransactionDetails()

CLASS UPI
    upiId
    amount
    merchantName
    
    METHOD Constructor(upiId, amount, merchantName)
        SET fields accordingly
    
    METHOD getTransactionDetails()
        RETURN "Payment of ₹" + amount + " via UPI successful for " + merchantName

CLASS Card
    cardNumber
    cvv
    expiryDate
    amount
    
    METHOD Constructor(cardNumber, cvv, expiryDate, amount)
        SET fields accordingly
    
    METHOD getTransactionDetails()
        RETURN "Payment of ₹" + amount + " via Card ending " + last4(cardNumber) + " successful."

CLASS NetBanking
    bankName
    accountNumber
    amount
    
    METHOD Constructor(bankName, accountNumber, amount)
        SET fields accordingly
    
    METHOD getTransactionDetails()
        RETURN "Payment of ₹" + amount + " via NetBanking (" + bankName + ") successful."

CLASS Student:
    name
    marks

FUNCTION compare(student1, student2):
    IF student1.marks > student2.marks:
        RETURN true
    ELSE IF student1.marks < student2.marks:
        RETURN false
    ELSE:
        RETURN student1.name < student2.name

FUNCTION merge(leftList, rightList):
    result = empty list
    i = 0
    j = 0

    WHILE i < length(leftList) AND j < length(rightList):
        IF compare(leftList[i], rightList[j]) == true:
            ADD leftList[i] TO result
            i = i + 1
        ELSE:
            ADD rightList[j] TO result
            j = j + 1

    WHILE i < length(leftList):
        ADD leftList[i] TO result
        i = i + 1

    WHILE j < length(rightList):
        ADD rightList[j] TO result
        j = j + 1

    RETURN result

FUNCTION mergeSort(studentList):
    IF length(studentList) <= 1:
        RETURN studentList

    mid = length(studentList) / 2
    left = mergeSort(studentList[0 .. mid-1])
    right = mergeSort(studentList[mid .. end])

    RETURN merge(left, right)

MAIN PROGRAM:
    studentList = [
        ("Amit", 92),
        ("Bhavna", 92),
        ("Kiran", 85),
        ("Dinesh", 78),
        ("Charu", 92)
    ]

    sortedList = mergeSort(studentList)

    PRINT "Final Rank List:"
    FOR each student IN sortedList:
        PRINT student.name + " - " + student.marks
